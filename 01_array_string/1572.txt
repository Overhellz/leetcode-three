class Solution {
    public int diagonalSum(int[][] mat) {
        int n = mat.length - 1;
        int m = mat[0].length - 1;
        int sum = 0;

        for(int i = 0; i <= n; i++) {
            for(int j = 0; j <= m; j++) {
                if (i == j || i == n - j) {
                    sum += mat[i][j];
                }
            }
        }
        return sum;
    }
}
