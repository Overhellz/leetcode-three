class Solution {
    public int findLengthOfLCIS(int[] nums) {
        int cur = 1;
        int ans = 1;
        for(int i = 1; i < nums.length; i++) {
            if (nums[i] > nums[i-1]) {
                cur++;
                ans = Math.max(ans, cur);
            } else {
                cur = 1;
            }
        }
        return ans;
    }
}
