class Solution {
    public String addBinary(String a, String b) {
        int posA = a.length() - 1;
        int posB = b.length() - 1;
        int sum = 0;

        StringBuilder result = new StringBuilder();

        while(posA >= 0 && posB >= 0) {
            if (a.charAt(posA) == '0' && b.charAt(posB) == '0' && sum == 0) {
                result.append('0');
                sum = 0;
            } else if (a.charAt(posA) == '1' && b.charAt(posB) == '1' && sum == 0) {
                result.append('0');
                sum = 1;
            } else if (a.charAt(posA) == '1' && b.charAt(posB) == '1' && sum == 1) {
                result.append('1');
                sum = 1;
            } else if (a.charAt(posA) == '1' && b.charAt(posB) == '0' && sum == 1) {
                result.append('0');
                sum = 1;
            } else if (a.charAt(posA) == '0' && b.charAt(posB) == '1' && sum == 1) {
                result.append('0');
                sum = 1;
            }
            posA--;
            posB--;
        }

        while(posA >= 0) {
            if (a.charAt(posA) == '0' && sum == 0) {
                result.append('0');
                sum = 0;
            } else if (a.charAt(posA) == '1' && sum == 0) {
                result.append('1');
                sum = 0;
            } else if (a.charAt(posA) == '0' && sum == 1) {
                result.append('1');
                sum = 0;
            }
            posA--;
        }
        while(posB >= 0) {
            if (b.charAt(posB) == '0' && sum == 0) {
                result.append('0');
                sum = 0;
            } else if (b.charAt(posB) == '1' && sum == 0) {
                result.append('1');
                sum = 0;
            } else if (b.charAt(posB) == '0' && sum == 1) {
                result.append('1');
                sum = 0;
            }
            posB--;
        }

        if (sum == 1) {
            result.append('1');
        }

        result.reversed();
        return result.toString();
    }
}
