class Solution {
    public int romanToInt(String s) {
        int result = 0;
        int n = s.length();
        for(int i = 0; i < n; i++) {
            if (i + 1 < n && value(s.charAt(i)) < value(s.charAt(i + 1))) {
                result -= value(s.charAt(i));
            } else {
                result += value(s.charAt(i));
            }
        }
        return result;
    }

    private int value(char ch) {
        return switch(ch) {
            case 'I' -> 1;
            case 'V' -> 5;
            case 'X' -> 10;
            case 'L' -> 50;
            case 'C' -> 100;
            case 'D' -> 500;
            case 'M' -> 1000;
            default -> throw new IllegalArgumentException("bad char: " + c);
        }
    }
}
