class Solution {
    public int removeDuplicates(int[] nums) {
        int n = 1;
        int pos = 0;
        int num = nums[0];

        for(int i = 1; i < nums.length; i++) {
            if (n == 1) {
                if (nums[i] == nums[pos]) {
                    n = 2;
                    pos++;
                } else {
                    n = 1;
                    pos = i;
                }
            } else {
                if (n == 2) {
                    if (nums[i] == nums[pos]) {
                        continue;
                    } else {
                        pos++;
                        n = 1;
                    }
                }
                continue;
            }
        }
        return pos + 1;
    }
}
