class Solution {
    public int lengthOfLastWord(String s) {
        int last = s.length() - 1;
        for(int i = s.length() - 1; i >= 0; i--) {
            if (s.charAt(i) != ' ') {
                last = i;
                break;
            }
        }

        int first = 0;
        for(int i = last - 1; i >= 0; i--) {
            if (s.charAt(i) == ' ') {
                first = i + 1;
                break;
            }
        }
        return last - first + 1;
    }
}
